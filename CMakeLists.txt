cmake_minimum_required(VERSION 3.24)
project(MQL5GPULibrary_LSTM VERSION 3.0.0 LANGUAGES CXX)

include(CTest)
option(DN_BUILD_CUDA "Also build the version 2.3 Windows CUDA DLL" OFF)
set(DN3_ORT_ROOT "" CACHE PATH "Extracted ONNX Runtime C/C++ SDK (optional)")
find_package(Threads REQUIRED)
add_library(DNForecast SHARED src/forecast_runtime.cpp src/forecast_model.cpp src/forecast_legacy.cpp)
target_compile_features(DNForecast PRIVATE cxx_std_17)
target_compile_definitions(DNForecast PRIVATE DN3_BUILD)
target_include_directories(DNForecast PUBLIC include)
target_link_libraries(DNForecast PRIVATE Threads::Threads)
set_target_properties(DNForecast PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(MSVC)
    set_property(TARGET DNForecast PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(DN3_ORT_ROOT)
    find_path(DN3_ORT_INCLUDE onnxruntime_cxx_api.h PATHS "${DN3_ORT_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
    find_library(DN3_ORT_LIBRARY onnxruntime PATHS "${DN3_ORT_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
    target_include_directories(DNForecast PRIVATE "${DN3_ORT_INCLUDE}")
    target_link_libraries(DNForecast PRIVATE "${DN3_ORT_LIBRARY}")
    target_compile_definitions(DNForecast PRIVATE DN3_ORT)
    if(WIN32)
        file(GLOB DN3_ORT_DLLS "${DN3_ORT_ROOT}/lib/*.dll")
        add_custom_command(TARGET DNForecast POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
            ${DN3_ORT_DLLS} "$<TARGET_FILE_DIR:DNForecast>")
    endif()
endif()

if(BUILD_TESTING)
    enable_language(C)
    add_executable(forecast_example examples/forecast.c)
    target_link_libraries(forecast_example PRIVATE DNForecast)
    target_compile_features(forecast_example PRIVATE c_std_11)
    add_test(NAME forecast_c COMMAND forecast_example)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_executable(validation_tests tests/validation_tests.cpp)
    target_include_directories(validation_tests PRIVATE include)
    target_compile_features(validation_tests PRIVATE cxx_std_17)
    add_test(NAME validation COMMAND validation_tests)
    add_test(NAME api_contract COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/check_api.py")
    add_test(NAME financial_series COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_financial_series.py")
    add_test(NAME forecast_runtime COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_forecast.py" "$<TARGET_FILE:DNForecast>")
endif()

if(DN_BUILD_CUDA)
    if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "The MT5 DLL requires 64-bit Windows. Use DN_BUILD_CUDA=OFF for portable tests.")
    endif()
    enable_language(CUDA)
    find_package(CUDAToolkit REQUIRED)
    add_library(MQL5GPULibrary_LSTM SHARED kernel.cu)
    set_property(SOURCE kernel.cu PROPERTY OBJECT_DEPENDS
        "${CMAKE_CURRENT_SOURCE_DIR}/include/model_validation.h;${CMAKE_CURRENT_SOURCE_DIR}/include/checkpoint.h;${CMAKE_CURRENT_SOURCE_DIR}/include/MQL5GPULibrary.h;${CMAKE_CURRENT_SOURCE_DIR}/include/topology.h;${CMAKE_CURRENT_SOURCE_DIR}/src/graph_layers.cuh")
    # NVIDIA's VS custom build does not reliably discover nested local headers.
    set_property(SOURCE kernel.cu PROPERTY VS_SETTINGS
        "AdditionalDependencies=${CMAKE_CURRENT_SOURCE_DIR}/include/model_validation.h$<SEMICOLON>${CMAKE_CURRENT_SOURCE_DIR}/include/checkpoint.h$<SEMICOLON>${CMAKE_CURRENT_SOURCE_DIR}/include/MQL5GPULibrary.h$<SEMICOLON>${CMAKE_CURRENT_SOURCE_DIR}/include/topology.h$<SEMICOLON>${CMAKE_CURRENT_SOURCE_DIR}/src/graph_layers.cuh")
    target_compile_features(MQL5GPULibrary_LSTM PRIVATE cxx_std_17 cuda_std_17)
    target_link_libraries(MQL5GPULibrary_LSTM PRIVATE CUDA::cudart CUDA::cublas CUDA::curand)
    set_target_properties(MQL5GPULibrary_LSTM PROPERTIES CUDA_RUNTIME_LIBRARY Shared)
    if(BUILD_TESTING)
        add_test(NAME gpu_api COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_dll.py"
            "$<TARGET_FILE:MQL5GPULibrary_LSTM>" "${CUDAToolkit_BIN_DIR}")
        set_tests_properties(gpu_api PROPERTIES TIMEOUT 180 LABELS gpu)
        add_test(NAME forecast_legacy COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_forecast_legacy.py"
            "$<TARGET_FILE:DNForecast>" "$<TARGET_FILE:MQL5GPULibrary_LSTM>" "${CUDAToolkit_BIN_DIR}")
    endif()
endif()
